cmake_minimum_required(VERSION 3.16)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    return()
endif()
project(selective_lim LANGUAGES CXX)

set(SOC_VERSION "Ascend910_9391" CACHE STRING "Target Ascend SoC")
set(RUN_MODE "npu" CACHE STRING "AscendC run mode")
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN installation")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
include("$ENV{ASCEND_HOME_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
ascendc_library(selective_lim SHARED selective_lim.cpp)
target_sources(selective_lim PRIVATE host_gather.cpp)
target_include_directories(selective_lim PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(selective_lim PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(selective_lim PRIVATE ascendcl)
